② VectorIndex クラスの完全実装

import math
from typing import Optional, Any, List, Dict, Tuple


class VectorIndex:
    def __init__(self, distance_metric: str = "cosine", embedding_fn=None):
        self.vectors: List[List[float]] = []
        self.documents: List[Dict[str, Any]] = []
        self._vector_dim: Optional[int] = None
        if distance_metric not in ["cosine", "euclidean"]:
            raise ValueError("distance_metric must be 'cosine' or 'euclidean'")
        self._distance_metric = distance_metric
        self._embedding_fn = embedding_fn

    def add_document(self, document: Dict[str, Any]):
        """1件追加: contentからembedding生成 → 保存"""
        if not self._embedding_fn:
            raise ValueError("Embedding function not provided")
        if "content" not in document:
            raise ValueError("Document must contain 'content' key")

        vector = self._embedding_fn(document["content"])
        self.add_vector(vector=vector, document=document)

    def add_vector(self, vector, document):
        """次元一貫性チェック + 追加"""
        if not self.vectors:
            self._vector_dim = len(vector)
        elif len(vector) != self._vector_dim:
            raise ValueError(f"Inconsistent vector dimension")
        self.vectors.append(list(vector))
        self.documents.append(document)

    def search(self, query, k: int = 1) -> List[Tuple[Dict, float]]:
        """類似検索: クエリ → embedding → 全 vector と距離 → top-k"""
        if not self.vectors:
            return []

        # クエリが string なら embedding 生成、リストならそのまま使う
        if isinstance(query, str):
            query_vector = self._embedding_fn(query)
        else:
            query_vector = query

        if len(query_vector) != self._vector_dim:
            raise ValueError("Query vector dimension mismatch")

        dist_func = self._cosine_distance if self._distance_metric == "cosine" else self._euclidean_distance

        distances = []
        for i, stored_vector in enumerate(self.vectors):
            distance = dist_func(query_vector, stored_vector)
            distances.append((distance, self.documents[i]))

        distances.sort(key=lambda item: item[0])  # 距離小さい順
        return [(doc, dist) for dist, doc in distances[:k]]

③ 距離計算メソッド (cosine と euclidean)

    def _cosine_distance(self, vec1, vec2) -> float:
        """Cosine距離: 1 - (内積 / (大きさ1 × 大きさ2))"""
        mag1 = self._magnitude(vec1)
        mag2 = self._magnitude(vec2)

        if mag1 == 0 and mag2 == 0:
            return 0.0
        elif mag1 == 0 or mag2 == 0:
            return 1.0

        dot_prod = self._dot_product(vec1, vec2)
        cosine_similarity = dot_prod / (mag1 * mag2)
        cosine_similarity = max(-1.0, min(1.0, cosine_similarity))  # 数値誤差クランプ

        return 1.0 - cosine_similarity

    def _euclidean_distance(self, vec1, vec2) -> float:
        """ユークリッド距離: sqrt(Σ(p-q)²)"""
        return math.sqrt(sum((p - q) ** 2 for p, q in zip(vec1, vec2)))

    def _dot_product(self, vec1, vec2) -> float:
        return sum(p * q for p, q in zip(vec1, vec2))

    def _magnitude(self, vec) -> float:
        return math.sqrt(sum(x * x for x in vec))
cosine距離の使い分け: 距離 = 0 (完全一致) 〜 2 (正反対)。検索では 小さい順 にtop-k を取る。

④ 全フロー: report.md → 質問 → 検索 → 回答

# 1. report.md 読み込み
with open("./report.md", "r") as f:
    text = f.read()

# 2. チャンク分割
chunks = chunk_by_section(text)

# 3. VectorIndex 初期化 (embedding_fn を渡す)
store = VectorIndex(
    distance_metric="cosine",
    embedding_fn=lambda t: generate_embedding(t, input_type="document")
)

# 4. 全チャンクを add
for chunk in chunks:
    store.add_document({"content": chunk})

print(store)  # VectorIndex(count=15, dim=1024, metric='cosine', has_embedding_fn='Yes')

# 5. ユーザー質問で検索
query = "Anthropicの2024年の主な進捗は何ですか?"
results = store.search(query, k=2)  # top 2 関連chunk

for doc, distance in results:
    print(f"distance={distance:.4f}")
    print(f"content={doc['content'][:200]}...")
    print("---")

client = anthropic.Anthropic()

context_text = "\n---\n".join([doc["content"] for doc, _ in results])

上記の参考資料に基づいて、以下の質問に答えてください。

{query}
</質問>

    }]

⑥ バルク処理版 (大量データ向け)

def add_documents(self, documents: List[Dict[str, Any]], batch_size: int = 50):
    """大量データ用バルク処理 (Voyage AI レート制限回避)"""
    for i in range(0, len(documents), batch_size):
        batch = documents[i:i+batch_size]
        contents = [d["content"] for d in batch]

        # Voyage AI バッチembed (1回のAPI呼び出しで複数件)
        result = self._embedding_fn(contents)  # List[List[float]]

        for vec, doc in zip(result, batch):
            self.add_vector(vec, doc)

        time.sleep(0.5)  # レート制限回避
本ipynb での注意 (cell-7コメント): 「rate limiting errors from VoyageAI を回避するため、bulk operation に変換した」。100万件のチャンクは1件ずつ処理せず、必ず batch + sleep。

⑪ 理解度チェック

Q1. VectorIndex に必要な4つの内部状態は?

① self.vectors (ベクトルのリスト) ② self.documents (元chunk) ③ self._vector_dim (次元数・一貫性チェック用) ④ self._distance_metric (cosine or euclidean)

Q2. なぜ次元一貫性チェックが必要?

異なる embedding model で生成したベクトル (例: voyage-3-large 1024次元 vs OpenAI 1536次元) が混ざると距離計算が壊れる。最初に self._vector_dim を記録し、以降は一致確認。

Q3. search で距離が小さい順にソートする理由は?

cosine距離・euclidean距離はどちらも 0に近いほど似ている。
cosine類似度 (sim) なら大きい順だが、距離 (1 - sim) を使うので小さい順。検索意図は「最も似たchunkトップk」。

Q4. なぜバルク処理が必要?

1チャンクずつ API呼び出しすると Voyage AI レート制限 (3 RPM 無料枠) に引っかかる。
add_documents でバッチ化 + time.sleep(0.5) 挿入。本ipynb cell-7 にバルク版実装あり。
